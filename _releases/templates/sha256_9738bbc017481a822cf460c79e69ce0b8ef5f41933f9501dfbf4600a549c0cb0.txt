name: terragucci

on:
  push:
    branches:
      - main
  pull_request: {}
  workflow_dispatch: {}

permissions:
  contents: read

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: '${{ github.ref != ''refs/heads/main'' }}'

jobs:
  static-checks:
    name: static
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - name: Setup Node.js
        uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020
        with:
          node-version: '24'
          cache: npm
      - name: Install just
        run: |
          curl -fsSL -o /tmp/just.tar.gz https://github.com/casey/just/releases/download/1.36.0/just-1.36.0-$(uname -m)-unknown-linux-musl.tar.gz
          tar -xzf /tmp/just.tar.gz -C /tmp just
          sudo mv /tmp/just /usr/local/bin/just
          just --version
      - name: Setup Go
        uses: actions/setup-go@b7ad1dad31e06c5925ef5d2fc7ad053ef454303e
        with:
          go-version: '1.25.5'
          cache-dependency-path: terraform-provider-terragucci/go.sum
      - name: Install
        run: npm ci
      - name: Check
        run: just check-unit
      - name: Lint docs
        run: just lint-docs
  test:
    name: test (${{ matrix.shard }}/3)
    runs-on: ubuntu-latest
    timeout-minutes: 10
    strategy:
      fail-fast: false
      matrix:
        shard:
          - 1
          - 2
          - 3
    steps:
      - name: Checkout
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - name: Setup Node.js
        uses: actions/setup-node@820762786026740c76f36085b0efc47a31fe5020
        with:
          node-version: '24'
          cache: npm
      - name: Install just
        run: |
          curl -fsSL -o /tmp/just.tar.gz https://github.com/casey/just/releases/download/1.36.0/just-1.36.0-$(uname -m)-unknown-linux-musl.tar.gz
          tar -xzf /tmp/just.tar.gz -C /tmp just
          sudo mv /tmp/just /usr/local/bin/just
          just --version
      - name: Install OpenTofu
        run: |
          arch=$(uname -m); case "$arch" in x86_64) arch=amd64 ;; aarch64) arch=arm64 ;; esac
          curl -fsSL -o /tmp/tofu.tar.gz https://github.com/opentofu/opentofu/releases/download/v1.13.1/tofu_1.13.1_linux_$arch.tar.gz
          tar -xzf /tmp/tofu.tar.gz -C /tmp tofu
          sudo mv /tmp/tofu /usr/local/bin/tofu
          tofu version
      - name: Install Terragrunt
        run: |
          arch=$(uname -m); case "$arch" in x86_64) arch=amd64 ;; aarch64) arch=arm64 ;; esac
          cd /tmp
          curl -fsSL -O https://github.com/gruntwork-io/terragrunt/releases/download/v1.2.0/terragrunt_linux_$arch
          curl -fsSL -O https://github.com/gruntwork-io/terragrunt/releases/download/v1.2.0/SHA256SUMS
          grep " terragrunt_linux_$arch\$" SHA256SUMS | sha256sum -c -
          sudo install -D -m 0755 terragrunt_linux_$arch /opt/terragrunt/terragrunt
          /opt/terragrunt/terragrunt --version
      - name: Install
        run: npm ci
      - name: Test
        run: just test-shard ${{ matrix.shard }} 3
        env:
          TERRAGUCCI_TEST_TERRAGRUNT: /opt/terragrunt/terragrunt
